# 薄弱知识点推荐模块重构方案三：学习状态机 + 独立推荐引擎

> 原则：每步可独立验证，按阶段提交，不破坏现有功能。

---

## 总体架构变化

```
旧架构：
  weak_points: JSON[]  (StudentProfile 上的一个字段，多处追加，无上限)

新架构：
  WeakPoint 表 (独立模型，状态机管理生命周期)
  RecommendationService (独立推荐引擎，SM-2 间隔重复算法)
  weak_points 字段保留用于兼容旧代码，逐步迁移
```

状态机流转：
```
active ──[做题正确率上升]──► reviewing ──[连续答对3次]──► mastered ──[手动归档]──► archived
  ▲                                ▼
  └──────[做题错误率上升]──────────┘
```

## 第一阶段：数据层（预计 0.5 天）

### 1.1 新建 WeakPoint 模型

新建文件 `backend/models/weak_point.py`：

```python
from sqlalchemy import Column, Integer, String, Float, DateTime, JSON, ForeignKey
from datetime import datetime, timezone
from core.database import Base

class WeakPoint(Base):
    __tablename__ = "weak_points"

    id = Column(Integer, primary_key=True, autoincrement=True)
    user_id = Column(String, index=True, nullable=False)
    name = Column(String, nullable=False)          # 知识点名称

    # 状态机：active / reviewing / mastered / archived
    status = Column(String, default="active", nullable=False)

    mastery_score = Column(Float, default=0.0)     # 0.0 ~ 1.0，答题正确率累计
    quiz_count = Column(Integer, default=0)        # 总答题次数
    correct_count = Column(Integer, default=0)     # 答对次数

    # SM-2 间隔重复字段
    ease_factor = Column(Float, default=2.5)       # 难度系数，初始 2.5
    interval_days = Column(Integer, default=1)     # 下次复习间隔（天）
    next_review_at = Column(DateTime, nullable=True)  # 下次复习时间

    # 关联的资源 id 列表
    related_resource_ids = Column(JSON, default=list)

    first_seen_at = Column(DateTime, default=lambda: datetime.now(timezone.utc))
    last_quizzed_at = Column(DateTime, nullable=True)
    updated_at = Column(DateTime, default=lambda: datetime.now(timezone.utc),
                        onupdate=lambda: datetime.now(timezone.utc))
```

```python
"""一次性迁移脚本：将旧 weak_points 字段迁移到 WeakPoint 表"""
from core.database import SessionLocal
from models.student import StudentProfile
from models.weak_point import WeakPoint
from datetime import datetime, timezone

def migrate():
    db = SessionLocal()
    try:
        profiles = db.query(StudentProfile).all()
        for p in profiles:
            if not p.weak_points:
                continue
            for name in p.weak_points:
                exists = db.query(WeakPoint).filter(
                    WeakPoint.user_id == p.user_id,
                    WeakPoint.name == name
                ).first()
                if not exists:
                    db.add(WeakPoint(user_id=p.user_id, name=name))
        db.commit()
        print("迁移完成")
    finally:
        db.close()

if __name__ == "__main__":
    migrate()
```

## 第二阶段：推荐引擎服务（预计 1 天）

### 2.1 SM-2 算法核心

新建 `backend/services/recommendation_service.py`：

```python
"""
推荐引擎服务：基于 SM-2 间隔重复算法管理薄弱知识点生命周期。

SM-2 规则：
  quality 0~1（本次答题质量，正确率）
  如果 quality >= 0.6：
    interval = 上次 interval * ease_factor
    ease_factor = ease_factor + 0.1 - (1-quality) * 0.8
  否则：
    interval = 1  （重置，明天再考）
    ease_factor 不变
  ease_factor 最小值 = 1.3
  interval 最大值 = 180 天
"""
from datetime import datetime, timezone, timedelta
from core.database import SessionLocal
from models.weak_point import WeakPoint

MIN_EASE = 1.3
MAX_INTERVAL = 180
MASTERY_THRESHOLD = 0.75    # mastery_score 超过此值且 quiz_count >= 5 → mastered
REVIEWING_THRESHOLD = 0.5   # mastery_score 超过此值 → reviewing


def _sm2_update(wp: WeakPoint, quality: float) -> None:
    """就地更新 SM-2 字段，quality 为本次答题正确率（0.0~1.0）"""
    if quality >= 0.6:
        if wp.quiz_count == 1:
            wp.interval_days = 1
        elif wp.quiz_count == 2:
            wp.interval_days = 6
        else:
            wp.interval_days = max(1, int(wp.interval_days * wp.ease_factor))
        wp.ease_factor = max(MIN_EASE, wp.ease_factor + 0.1 - (1 - quality) * 0.8)
    else:
        wp.interval_days = 1

    wp.interval_days = min(wp.interval_days, MAX_INTERVAL)
    wp.next_review_at = datetime.now(timezone.utc) + timedelta(days=wp.interval_days)


def _update_status(wp: WeakPoint) -> None:
    """根据 mastery_score 和 quiz_count 自动推进状态"""
    if wp.status == "archived":
        return
    score = wp.mastery_score
    if score >= MASTERY_THRESHOLD and wp.quiz_count >= 5:
        wp.status = "mastered"
    elif score >= REVIEWING_THRESHOLD:
        wp.status = "reviewing"
    else:
        wp.status = "active"


def record_quiz_result(user_id: str, weak_point_name: str, correct: bool) -> None:
    """
    记录一次答题结果，更新对应 WeakPoint 的状态和 SM-2 字段。
    如果该知识点不存在则自动创建。
    """
    db = SessionLocal()
    try:
        wp = db.query(WeakPoint).filter(
            WeakPoint.user_id == user_id,
            WeakPoint.name == weak_point_name,
            WeakPoint.status != "archived"
        ).first()

        if not wp:
            wp = WeakPoint(user_id=user_id, name=weak_point_name)
            db.add(wp)

        wp.quiz_count += 1
        if correct:
            wp.correct_count += 1
        wp.mastery_score = wp.correct_count / wp.quiz_count
        wp.last_quizzed_at = datetime.now(timezone.utc)

        quality = 1.0 if correct else 0.0
        _sm2_update(wp, quality)
        _update_status(wp)

        db.commit()
    finally:
        db.close()


def upsert_weak_point(user_id: str, name: str) -> None:
    """新增薄弱知识点（已存在则跳过）"""
    db = SessionLocal()
    try:
        exists = db.query(WeakPoint).filter(
            WeakPoint.user_id == user_id,
            WeakPoint.name == name
        ).first()
        if not exists:
            db.add(WeakPoint(user_id=user_id, name=name))
            db.commit()
    finally:
        db.close()


def upsert_weak_points_batch(user_id: str, names: list[str]) -> None:
    """批量新增薄弱知识点（语义去重：只做精确匹配，语义去重见 2.2）"""
    db = SessionLocal()
    try:
        existing = {
            wp.name for wp in db.query(WeakPoint.name).filter(
                WeakPoint.user_id == user_id
            ).all()
        }
        for name in names:
            if name not in existing:
                db.add(WeakPoint(user_id=user_id, name=name))
                existing.add(name)
        db.commit()
    finally:
        db.close()


def get_active_recommendations(user_id: str, limit: int = 10) -> list[WeakPoint]:
    """
    返回推荐列表（active + reviewing），按优先级排序：
      1. 到期需复习的（next_review_at <= now）
      2. 掌握度低的
      3. 最近未做题的
    """
    now = datetime.now(timezone.utc)
    db = SessionLocal()
    try:
        wps = db.query(WeakPoint).filter(
            WeakPoint.user_id == user_id,
            WeakPoint.status.in_(["active", "reviewing"])
        ).all()

        def sort_key(wp: WeakPoint):
            overdue = 1 if (wp.next_review_at and wp.next_review_at <= now) else 0
            return (-overdue, wp.mastery_score, wp.last_quizzed_at or datetime.min)

        wps.sort(key=sort_key)
        # 返回前 limit 条，但不关闭 session 前先转为 dict 防止 lazy load
        result = wps[:limit]
        db.expunge_all()
        return result
    finally:
        db.close()
```

### 2.2 LLM 语义去重（可选，增强项）

在 `recommendation_service.py` 中增加以下函数，仅在新增知识点时调用一次：

```python
async def deduplicate_with_llm(user_id: str, new_name: str) -> str:
    """
    检查 new_name 是否与已有知识点语义重叠。
    若重叠则返回已有知识点名称；否则返回 new_name 本身。
    """
    from core.llm_client import chat_completion
    db = SessionLocal()
    try:
        existing_names = [
            wp.name for wp in db.query(WeakPoint.name).filter(
                WeakPoint.user_id == user_id,
                WeakPoint.status != "archived"
            ).limit(30).all()
        ]
    finally:
        db.close()

    if not existing_names:
        return new_name

    prompt = f"""已有知识点列表：{existing_names}
新知识点：{new_name}

判断"新知识点"是否与列表中某个知识点表达的是同一概念（语义相同）。
若相同，返回列表中那个知识点的原文；若不同，返回"{new_name}"。
只返回一个字符串，不加任何解释。"""

    resp = await chat_completion([{"role": "user", "content": prompt}], temperature=0.0)
    return resp.choices[0].message.content.strip().strip('"')
```

## 第三阶段：API 路由（预计 0.5 天）

新建 `backend/api/routes/weak_point.py`：

```python
from fastapi import APIRouter, Query, HTTPException
from services.recommendation_service import (
    get_active_recommendations, upsert_weak_point, record_quiz_result
)
from core.database import SessionLocal
from models.weak_point import WeakPoint

router = APIRouter(prefix="/api/weak-points", tags=["weak_points"])


@router.get("")
def list_weak_points(user_id: str = Query(...), status: str = Query(None)):
    """获取薄弱知识点列表，可按 status 过滤"""
    db = SessionLocal()
    try:
        q = db.query(WeakPoint).filter(WeakPoint.user_id == user_id)
        if status:
            q = q.filter(WeakPoint.status == status)
        items = q.order_by(WeakPoint.mastery_score).all()
        return [_to_dict(w) for w in items]
    finally:
        db.close()


@router.get("/recommendations")
def get_recommendations(user_id: str = Query(...), limit: int = Query(10)):
    """获取推荐攻克列表（active+reviewing，按优先级排序）"""
    wps = get_active_recommendations(user_id, limit)
    return [_to_dict(w) for w in wps]


@router.patch("/{wp_id}/status")
def update_status(wp_id: int, status: str, user_id: str = Query(...)):
    """手动更新状态（mastered / archived / active）"""
    allowed = {"active", "reviewing", "mastered", "archived"}
    if status not in allowed:
        raise HTTPException(400, f"status 须为 {allowed} 之一")
    db = SessionLocal()
    try:
        wp = db.query(WeakPoint).filter(
            WeakPoint.id == wp_id, WeakPoint.user_id == user_id
        ).first()
        if not wp:
            raise HTTPException(404, "未找到")
        wp.status = status
        db.commit()
        return _to_dict(wp)
    finally:
        db.close()


@router.delete("/{wp_id}")
def delete_weak_point(wp_id: int, user_id: str = Query(...)):
    """删除（物理删除，或改用 archived 状态软删除）"""
    db = SessionLocal()
    try:
        wp = db.query(WeakPoint).filter(
            WeakPoint.id == wp_id, WeakPoint.user_id == user_id
        ).first()
        if not wp:
            raise HTTPException(404, "未找到")
        db.delete(wp)
        db.commit()
        return {"ok": True}
    finally:
        db.close()


def _to_dict(w: WeakPoint) -> dict:
    return {
        "id": w.id,
        "name": w.name,
        "status": w.status,
        "mastery_score": round(w.mastery_score, 2),
        "quiz_count": w.quiz_count,
        "interval_days": w.interval_days,
        "next_review_at": w.next_review_at.isoformat() if w.next_review_at else None,
        "last_quizzed_at": w.last_quizzed_at.isoformat() if w.last_quizzed_at else None,
        "related_resource_ids": w.related_resource_ids or [],
    }
```


## 第六阶段：联调与验证（预计 0.5 天）

按以下端到端流程逐步验证：

1. **新增弱点**：提交一道错题 → 查 `WeakPoint` 表出现新记录，`mastery_score=0.0`
2. **答对提升**：再答对同知识点 → `mastery_score` 上升，`interval_days` 增大
3. **状态推进**：连续答对 5 次 + `mastery_score >= 0.75` → 自动变为 `mastered`，推荐面板消失
4. **推荐排序**：手动将某条 `next_review_at` 改为过去时间 → 该条排在推荐列表最前
5. **手动归档**：点击 tag 旁的 `×` → 该条不再出现在推荐面板
