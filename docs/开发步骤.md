## 第四阶段：智能体基类 & 注册机制（预计 0.5 天）

### 4.1 智能体基类 — `backend/agents/base.py`

```python
from abc import ABC, abstractmethod
from typing import AsyncGenerator
from models.student import StudentProfile

class AgentState(dict):
    """所有智能体共享的状态，由 LangGraph 或手动编排传递"""
    user_id: str
    user_message: str
    profile: StudentProfile | None
    history: list

    def __getattr__(self, key):
        return self.get(key)

    def __setattr__(self, key, value):
        self[key] = value

class BaseAgent(ABC):
    name: str = ""
    description: str = ""

    @abstractmethod
    async def process(self, state: AgentState) -> AgentState:
        """处理请求并返回更新后的状态"""
        ...

    async def stream(self, state: AgentState) -> AsyncGenerator[str, None]:
        """流式处理（默认调用 process 后 yield 结果）"""
        result = await self.process(state)
        yield result.get("response", "")
```

### 4.2 智能体注册表 — `backend/agents/registry.py`

```python
_registry: dict[str, "BaseAgent"] = {}

def register(agent: "BaseAgent"):
    _registry[agent.name] = agent

def get_agent(name: str):
    if name not in _registry:
        raise AgentNotFoundError(name)
    return _registry[name]

def get_all_agents():
    return list(_registry.values())

def reset():
    _registry.clear()
```

### 4.3 对话路由服务 — `backend/services/chat_service.py`

```python
from core.llm_client import chat_completion
from agents.registry import get_agent
from core.sse import sse_stream
from fastapi.responses import StreamingResponse

INTENT_PROMPT = """根据用户消息判断应调用哪个智能体。
可选智能体：{agent_list}
只返回智能体名称，不要其他内容。"""

async def route_to_agent(user_id: str, message: str):
    # 1. 用轻量 LLM 做意图识别
    agents = get_all_agents()
    agent_desc = "\n".join([f"- {a.name}: {a.description}" for a in agents])

    intent_resp = await chat_completion([
        {"role": "system", "content": INTENT_PROMPT.format(agent_list=agent_desc)},
        {"role": "user", "content": message}
    ])
    agent_name = intent_resp.choices[0].message.content.strip()

    # 2. 交给对应智能体处理
    agent = get_agent(agent_name)
    state = AgentState(user_id=user_id, user_message=message)
    result = await agent.stream(state)

    return StreamingResponse(sse_stream(result), media_type="text/event-stream")
```

### 6.1 事件系统 — `backend/events/`

```python
# events/events.py
class Event:
    PROFILE_UPDATED = "profile.updated"
    EVALUATION_COMPLETED = "evaluation.completed"
    PATH_UPDATED = "path.updated"

# events/dispatcher.py
from collections import defaultdict
from typing import Callable, Coroutine

_listeners: dict[str, list[Callable]] = defaultdict(list)

def on(event: str):
    def decorator(func):
        _listeners[event].append(func)
        return func
    return decorator

async def emit(event: str, data: dict):
    for listener in _listeners.get(event, []):
        await listener(data)
```
